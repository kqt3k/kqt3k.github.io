    function migrateSettings(s) {
        // v9.3 → v9.4: cosmeticsIdentityOn разделяем на ник + аватар
        if (s.cosmeticsIdentityOn !== undefined && s.cosmeticsNicknameOn === undefined) {
            s.cosmeticsNicknameOn = !!s.cosmeticsIdentityOn;
            s.cosmeticsAvatarOn = !!s.cosmeticsIdentityOn;
            delete s.cosmeticsIdentityOn;
        }
        // persistence дефолт
        if (!s.persistence) s.persistence = {...DEFAULT_PERSISTENCE};
        for (const k in DEFAULT_PERSISTENCE) {
            if (s.persistence[k] === undefined) s.persistence[k] = DEFAULT_PERSISTENCE[k];
        }
        // остальные недостающие поля
        for (const k in DEFAULT_SETTINGS) {
            if (s[k] === undefined) s[k] = DEFAULT_SETTINGS[k];
        }
        return s;
    }

    function saveSettings(s) {
        try { localStorage.setItem(SETTINGS_KEY, JSON.stringify(s)); }

    function findAvatarContainers() {
        const candidates = new Set();
        document.querySelectorAll('[class*="avatar__part" i]').forEach(part => {
            let container = part.parentElement;
            let depth = 0;
            while (container && container !== document.body && depth < 5) {
                const rect = container.getBoundingClientRect();
                if (rect.width >= 24 && rect.width <= 200 && Math.abs(rect.width - rect.height) < 20) {
                    candidates.add(container); break;
                }
                container = container.parentElement; depth++;
            }
        });
        document.querySelectorAll('[class*="avatar" i], [class*="profile-pic" i], [data-testid*="avatar" i]').forEach(el => {
            if (/avatar__part/i.test(el.className || '')) return;
            const rect = el.getBoundingClientRect();
            if (rect.width >= 24 && rect.width <= 200 && Math.abs(rect.width - rect.height) < 20) candidates.add(el);
        });
        return Array.from(candidates);
    }

    function applyCustomAvatar(el, on) {
        if (!el) return;
        if (on && settings.customAvatarImg) {
            const src = settings.customAvatarImg;
            el.style.setProperty('background-image', `url("${src}")`, 'important');
            el.style.setProperty('background-size', 'cover', 'important');
            el.style.setProperty('background-position', 'center', 'important');
            el.style.setProperty('background-repeat', 'no-repeat', 'important');
            el.style.setProperty('border-radius', '50%', 'important');
            el.style.setProperty('overflow', 'hidden', 'important');
            el.querySelectorAll('*').forEach(child => child.style.setProperty('visibility', 'hidden', 'important'));
        } else {
            el.style.removeProperty('background-image');
            el.style.removeProperty('background-size');
            el.style.removeProperty('background-position');
            el.style.removeProperty('background-repeat');
            el.style.removeProperty('border-radius');
            el.style.removeProperty('overflow');
            el.querySelectorAll('*').forEach(child => child.style.removeProperty('visibility'));
        }
    }

    function replaceTextInNode(node, fromText, toText) {
        if (node.nodeType === Node.TEXT_NODE) {
            if (node.nodeValue.includes(fromText)) {
                node.nodeValue = node.nodeValue.replace(new RegExp(fromText, 'g'), toText);
            }
        } else if (node.nodeType === Node.ELEMENT_NODE) {
            if (node.tagName !== 'SCRIPT' && node.tagName !== 'STYLE' && node.tagName !== 'TEXTAREA') {
                for (let i = 0; i < node.childNodes.length; i++) replaceTextInNode(node.childNodes[i], fromText, toText);
            }
        }
    }

    function injectCosmetics() {
        const dropdown = document.querySelector('[data-testid="headbar-dropdown"]');

        if (dropdown && !window.__uchi_orig_name) {
            const nameEl = dropdown.querySelector('.sc-iJKPnr, div[class*="name" i]');
            if (nameEl && nameEl.textContent.trim() !== settings.customNickname) {
                window.__uchi_orig_name = nameEl.textContent.trim();
            }
        }

        // ---------- НИК ----------
        if (!settings.cosmeticsNicknameOn) {
            // Возвращаем оригинал, если он был заменён
            if (window.__uchi_orig_name) {
                replaceTextInNode(document.body, settings.customNickname, window.__uchi_orig_name);
                document.querySelectorAll('div').forEach(el => {
                    if (el.children.length === 0 && el.textContent.trim() === settings.customNickname) {
                        el.textContent = window.__uchi_orig_name;
                    }
                });
            }
            // Всё равно показываем имя (убираем анти-мерцание)
            if (dropdown && !dropdown.classList.contains('uchi-name-ready')) {
                dropdown.classList.add('uchi-name-ready');
            }
        } else {
            if (dropdown && settings.customNickname) {
                const nameEls = dropdown.querySelectorAll('.sc-iJKPnr, div[class*="name" i]');
                nameEls.forEach(el => {
                    if (el.textContent.trim() !== settings.customNickname) el.textContent = settings.customNickname;
                });
                if (!dropdown.classList.contains('uchi-name-ready')) dropdown.classList.add('uchi-name-ready');
            }
            if (window.__uchi_orig_name && settings.customNickname) {
                const firstName = window.__uchi_orig_name.split(' ')[0];
                if (firstName && firstName.length > 1) {
                    replaceTextInNode(document.body, firstName, settings.customNickname);
                    document.querySelectorAll('div').forEach(el => {
                        if (el.children.length === 0 && el.innerHTML.includes('<br>') && el.innerHTML.includes(firstName)) {
                            el.innerHTML = settings.customNickname;
                        }
                    });
                }
            }
        }

        // ---------- АВАТАРКА ----------
        if (settings.cosmeticsAvatarOn && settings.customAvatarImg) {
            document.documentElement.classList.add('uchi-replace-avatar');
            findAvatarContainers().forEach(el => applyCustomAvatar(el, true));
        } else {
            document.documentElement.classList.remove('uchi-replace-avatar');
            findAvatarContainers().forEach(el => applyCustomAvatar(el, false));
        }
    }

    let observerTimeout = null;
    function startCosmeticObserver() {
        if (!document.body) return;
        const observer = new MutationObserver(() => {
            if (observerTimeout) return;
            observerTimeout = setTimeout(() => { injectCosmetics(); observerTimeout = null; }, 10);
        });
        observer.observe(document.body, { childList: true, subtree: true });
    }

    function handleImageUpload(file, maxSize, format, callback) {
        const reader = new FileReader();
        reader.onload = () => {
            const img = new Image();
            img.onload = () => {
                let w = img.width, h = img.height;
                if (Math.max(w, h) > maxSize) {
                    const k = maxSize / Math.max(w, h);
                    w = Math.round(w * k); h = Math.round(h * k);
                }
                const canvas = document.createElement('canvas');
                canvas.width = w; canvas.height = h;
                canvas.getContext('2d').drawImage(img, 0, 0, w, h);
                callback(canvas.toDataURL(format, 0.85));
            };
            img.src = reader.result;
        };
        reader.readAsDataURL(file);
    }

        if (code.startsWith('Key')) return code.replace('Key', '');
        if (code.startsWith('Digit')) return code.replace('Digit', '');
        return code;
    }

    // ==================== Режимы и решатель ====================
    function renderMode() {
        const modeToggle = document.getElementById('uchi-mode-toggle');
        if (!modeToggle) return;
        modeToggle.classList.toggle('on', settings.modeOn);
        document.getElementById('uchi-infinite-toggle').classList.toggle('on', settings.infiniteOn);

        document.getElementById('uchi-solve-block').classList.toggle('hidden', !settings.modeOn);

    }
    function escapeAttr(s) { return String(s == null ? '' : s).replace(/"/g, '&quot;'); }

    let solverRunning = false, solverTimer = null, solverLimit = null, solverSolved = 0;
    function stopSolver() { solverRunning = false; if (solverTimer) clearTimeout(solverTimer); updateStatus(); }
    function tryStartSolver(limit) {
        if (!isCardPage()) { updateStatus(); return; }
        stopSolver();
        solverLimit = (typeof limit === 'number' && limit > 0) ? limit : null;
        solverSolved = 0;
        solveCards();
    }
    function solveCards() {
        if (solverRunning || !isCardPage() || !isCardReady()) {
            if (isCardPage() && !isCardReady()) solverTimer = setTimeout(solveCards, 300);
            return;
        }
        const isOld = typeof Card.Player._emitSignal === 'undefined';
        const sendEvent = (name, data) => isOld ? Card.Player.__score.tutor._sys_event(name, data) : Card.Player._emitSignal(name, data);
        const getScoreJSON = () => { const n = {}; Card.Player.__score.save(n); return n; };
        const solveCurrent = () => {
            if (Card.Player.__score.current + 1 <= Card.Player.__score.total) Card.Player.__score.current++;
            if (Card.Player.__score._index + 2 <= Card.Player.__score.total) Card.Player.__score._index += 2;
            else Card.Player.__score._index--;
            sendEvent('beads_exercise_finish_succ', { amount: Card.Player.__score.current, total: Card.Player.__score.total });
            if (isOld) sendEvent('$store', getScoreJSON()); else sendEvent('$store', { json: JSON.stringify(getScoreJSON()) });
        };
        const reportSolve = () => sendEvent('$lesson_finish');

        solverRunning = true; updateStatus();
        const step = () => {
            try {
                if (!solverRunning || !isCardPage()) { solverRunning = false; updateStatus(); return; }
                if (solverLimit !== null && solverSolved >= solverLimit) { solverRunning = false; updateStatus(); return; }
                solveCurrent(); solverSolved++;
                settings.statsSolved++; settings.statsBeads++;
                saveSettings(settings);
                const elSolved = document.getElementById('stat-solved'), elBeads = document.getElementById('stat-beads');
                if (elSolved) elSolved.textContent = settings.statsSolved;
                if (elBeads) elBeads.textContent = settings.statsBeads;
                updateStatus();
                if (Card.Player.__score.current >= Card.Player.__score.total) { reportSolve(); solverRunning = false; updateStatus(); return; }
                solverTimer = setTimeout(step, 100);
            } catch(e) { console.error('[UchiAuto]', e); solverRunning = false; updateStatus(); }
        };
        step();
    }
    function updateStatus() {
        const infEl = document.getElementById('uchi-infinite-status'), cntEl = document.getElementById('uchi-count-status'), btn = document.getElementById('uchi-bar-do'), btn1 = document.getElementById('uchi-bar-one');
        if (btn) btn.disabled = solverRunning;
        if (btn1) btn1.disabled = solverRunning;
        if (!isCardPage()) { if (infEl) infEl.textContent = 'не карточка'; if (cntEl) cntEl.textContent = 'не карточка'; return; }
        if (!isCardReady()) { if (infEl) infEl.textContent = 'ждём карточку…'; if (cntEl) cntEl.textContent = 'ждём карточку…'; return; }
        const total = Card.Player.__score.total, cur = Card.Player.__score.current;
        if (infEl) infEl.textContent = !settings.infiniteOn ? '' : (solverRunning ? `решаем (всего ${cur}/${total})` : `готово (${cur}/${total})`);
        if (cntEl) cntEl.textContent = (solverRunning && settings.modeOn) ? `решаем: ${solverSolved} / лимит ${solverLimit || '∞'} (${cur}/${total})` : `готово (${cur}/${total}) — введите N`;
    }
    function isCardPage() { return location.pathname.startsWith('/cards/'); }
    function isCardReady() { return typeof Card !== 'undefined' && Card.Player && Card.Player.__score; }

    // ==================== Глобальные хоткеи ====================
    window.addEventListener('keydown', (e) => {
        const active = document.activeElement;
        const tag = active ? active.tagName : '';
        if (tag === 'INPUT' || tag === 'TEXTAREA' || (active && active.isContentEditable)) return;

        if (e.code === settings.hotkeyMenu) {
            e.preventDefault();
            const panel = document.getElementById('uchi-auto-panel');
            if (!panel) return;
            settings.menuVisible = !settings.menuVisible; saveSettings(settings);
            panel.style.display = settings.menuVisible ? 'flex' : 'none';
        }
        else if (e.code === settings.hotkeyBg) {
            e.preventDefault();
            settings.cosmeticsPageBgOn = !settings.cosmeticsPageBgOn;
            saveSettings(settings); applyPageBg();
            const t = document.getElementById('toggle-pagebg');
            if (t) t.classList.toggle('on', settings.cosmeticsPageBgOn);
        }
        else if (e.code === settings.hotkeyBeads) {
            e.preventDefault();
            settings.cosmeticsBeadsOn = !settings.cosmeticsBeadsOn;
            saveSettings(settings); applyBeadImage();
            const t = document.getElementById('toggle-beads');
            if (t) t.classList.toggle('on', settings.cosmeticsBeadsOn);
        }
        else if (e.code === settings.hotkeyIdentity) {
            e.preventDefault();
            // хоткей "Анонимность" переключает оба сразу (ник + аватар)
            const newVal = !(settings.cosmeticsNicknameOn && settings.cosmeticsAvatarOn);
            settings.cosmeticsNicknameOn = newVal;
            settings.cosmeticsAvatarOn = newVal;
            saveSettings(settings); injectCosmetics();
            const tn = document.getElementById('toggle-nickname');
            const ta = document.getElementById('toggle-avatar');
            if (tn) tn.classList.toggle('on', newVal);
            if (ta) ta.classList.toggle('on', newVal);
        }
        else if (e.code === settings.hotkeySolve1) {
            e.preventDefault();
            tryStartSolver(1);
        }
    });

    // Навигация
    let lastPath = location.pathname;
    setInterval(() => {
        if (location.pathname !== lastPath) { lastPath = location.pathname; stopSolver(); }
        if (!document.getElementById('uchi-auto-panel') && document.body) createPanel();
        if (solverRunning) updateStatus();
    }, 800);

    // Запуск
    function init() {
        applyPersistence();     // сначала применяем правила персистентности
        injectStyles();
        applyBeadImage();
        applyPageBg();
        createPanel();          // внутри вызовет applyMenuBg
        injectCosmetics();
        startCosmeticObserver();
        console.log('[UchiAuto] v9.4 загружена. Персистентность + раздельные тумблеры.');
    }

    if (document.body) init();
    else window.addEventListener('DOMContentLoaded', init);
})();
